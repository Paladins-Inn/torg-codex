== Authentifizierungsfluss

[plantuml, drivethru-auth-flow]
----
@startuml drivethru-auth-flow
skinparam defaultFontSize 11

actor        "HTTP Client"               as client
participant  "ApiKeyAuthenticationFilter" as filter  <<OncePerRequestFilter>>
participant  "ProviderManager"            as mgr     <<AuthenticationManager>>
participant  "DriveThruAuthenticationProvider" as prov <<AuthenticationProvider>>
participant  "DriveThruRPGService"        as svc     <<@Service, @Cacheable>>
participant  "DriveThruRPGClient"         as http    <<RestClient Proxy>>
boundary     "DriveThruRPG API"           as api

client ->> filter : HTTP\n Authorization: ApiKey <key>
activate filter

filter ->> mgr    : authenticate(\n  unauthenticated(apiKey))
activate mgr

mgr ->> prov      : authenticate(token)
activate prov

prov ->> svc      : getToken(apiKey)
activate svc

svc ->> http      : getToken("Bearer " + apiKey)
activate http
http ->> api      : POST /token
api -->> http     : {access_token, customers_id, expires}
http -->> svc     :
deactivate http

svc -->> prov     : DrivethruToken
deactivate svc

prov -->> mgr     : ApiKeyAuthenticationToken\n  .authenticated(userDetails)
deactivate prov

mgr -->> filter   :
deactivate mgr

filter ->> filter : SecurityContextHolder.set(auth)
filter -->> client : 200 + request continues
deactivate filter

@enduml
----

== Klassendiagramm

[plantuml, drivethru-security-classes]
----
@startuml drivethru-security-classes
skinparam defaultFontSize 11
skinparam classAttributeIconSize 0

package "de.paladinsinn.drivethru.security" {

  class DriveThruSecurityConfig <<@Configuration\n@EnableWebSecurity\n@EnableCaching>> {
    + authenticationManager(provider) : AuthenticationManager
    + securityFilterChain(http, mgr)  : SecurityFilterChain
  }

  class ApiKeyAuthenticationFilter <<OncePerRequestFilter>> {
    + {static} API_KEY_PREFIX : String = "ApiKey "
    - authenticationManager : AuthenticationManager
    # doFilterInternal(req, res, chain) : void
  }

  class ApiKeyAuthenticationToken <<AbstractAuthenticationToken>> {
    - principal : Object
    - credentials : Object
    + {static} unauthenticated(apiKey : String) : ApiKeyAuthenticationToken
    + {static} authenticated(user : DriveThruUserDetails) : ApiKeyAuthenticationToken
    + getPrincipal() : Object
    + getCredentials() : Object
  }

  class DriveThruAuthenticationProvider <<@Component\nAuthenticationProvider>> {
    - driveThruRPGService : DriveThruRPGService
    + authenticate(auth : Authentication) : Authentication
    + supports(cls : Class<?>) : boolean
  }

  class DriveThruUserDetails <<UserDetails>> {
    - token : DrivethruToken
    + getUsername() : String        ' customerId
    + getPassword() : String        ' null
    + getAuthorities() : Collection
    + getCustomerId() : String
    + getBearerToken() : String
    + getToken() : DrivethruToken
  }

  class DriveThruUserService <<@Service>> {
    - driveThruRPGService : DriveThruRPGService
    + getCurrentUser() : Optional<DriveThruUserDetails>
    + requireCurrentUser() : DriveThruUserDetails
    + getOwnedProductIds() : List<Integer>
    + getOwnedProducts(page, pageSize, archived) : List<OwnedProduct>
  }
}

package "de.paladinsinn.drivethru.token" {
  class DrivethruToken <<DriveThruResource>> {
    - accessToken : String
    - customerId  : String
    - expireTime  : LocalDateTime
    - serverTime  : LocalDateTime
    - localTime   : LocalDateTime
    - expires     : Long
    + getBearerToken() : String
  }
}

DriveThruSecurityConfig    ..>  ApiKeyAuthenticationFilter      : creates
DriveThruSecurityConfig    ..>  DriveThruAuthenticationProvider : uses
ApiKeyAuthenticationFilter ..>  ApiKeyAuthenticationToken       : creates unauthenticated
ApiKeyAuthenticationFilter -->  ApiKeyAuthenticationToken       : sets authenticated result
DriveThruAuthenticationProvider -->  DriveThruUserDetails       : creates
DriveThruAuthenticationProvider ..>  ApiKeyAuthenticationToken  : returns authenticated
DriveThruUserDetails        *--  DrivethruToken                 : wraps
DriveThruUserService        ..>  DriveThruUserDetails           : reads from SecurityContext

@enduml
