== Verarbeitungspipeline

[plantuml, markup-pipeline]
----
@startuml markup-pipeline
skinparam defaultFontSize 11

|TorgEntity|
start
:render(rawText);
note right: aufgerufen von getText(),\ngetWorldLaws() etc.

|Censor|
:apply(rawText);
:ConditionalBlockProcessor\n→ <IF:product>…</IF> auflösen;
note right
  Produkte aus DriveThruUserDetails
  des aktuellen Nutzers prüfen.
  Nicht freigeschaltete Blöcke
  werden entfernt.
end note

:EntityReferenceProcessor\n→ <perk:slug> etc. ersetzen;
note right
  Slug wird in entsprechenden
  Repository-Link aufgelöst.
end note

:GameTokenProcessor\n→ Spieltoken ersetzen;
note right
  z.B. Würfelsymbole → HTML-Icons
end note

:MarkdownProcessor\n→ CommonMark → HTML;

|TorgEntity|
:HTML zurückgeben;
stop

@enduml
----

package "de.paladinsinn.torg.codex.data.markup" {

  class TorgMarkupService <<@Service>> {
    - processors : List<MarkupProcessor>
    + process(raw : String, products : Set<String>) : String
  }

  interface MarkupProcessor {
    + process(text : String, context : MarkupContext) : String
  }

  class Censor <<@Component>> {
    - products : Set<String>
    + apply(rawText : String) : String
  }

  class ConditionalBlockProcessor <<MarkupProcessor>> {
    + process(text, ctx) : String
    ' Syntax: <IF:product-slug>…</IF>
  }

  class EntityReferenceProcessor <<MarkupProcessor>> {
    + process(text, ctx) : String
    ' Syntax: <type:slug>
  }

  class GameTokenProcessor <<MarkupProcessor>> {
    - registry : GameTokenRegistry
    + process(text, ctx) : String
  }

  class GameTokenRegistry <<@Component>> {
    - tokens : Map<String, String>
    + resolve(token : String) : String
  }

  class MarkdownProcessor <<MarkupProcessor>> {
    + process(text, ctx) : String
    ' CommonMark → HTML
  }

  class EntityType <<enum>> {
    PERK
    ITEM
    SPELL
    MIRACLE
    POWER
    THREAT
    COSM
    SHARD
  }
}

package "de.paladinsinn.torg.codex.data.model" {
  abstract class TorgEntity <<@MappedSuperclass>> {
    # censor : Censor
    # render(rawText : String) : String
    + withCensor(censor : Censor) : T
  }
}

TorgMarkupService   o--  Censor
Censor              ..>  ConditionalBlockProcessor : uses
Censor              ..>  EntityReferenceProcessor  : uses
Censor              ..>  GameTokenProcessor        : uses
Censor              ..>  MarkdownProcessor         : uses
GameTokenProcessor  o--  GameTokenRegistry
EntityReferenceProcessor ..> EntityType            : uses

TorgEntity          ..>  Censor : render() calls censor.apply()
ConditionalBlockProcessor ..|>  MarkupProcessor
EntityReferenceProcessor  ..|>  MarkupProcessor
GameTokenProcessor        ..|>  MarkupProcessor
MarkdownProcessor         ..|>  MarkupProcessor

@enduml
