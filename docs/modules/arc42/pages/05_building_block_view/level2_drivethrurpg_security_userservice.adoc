== Authentifizierungsfluss

[plantuml, drivethru-auth-flow]
----
@startuml drivethru-auth-flow
skinparam defaultFontSize 11

actor        "HTTP Client"               as client
participant  "ApiKeyAuthenticationFilter" as filter  <<OncePerRequestFilter>>
participant  "ProviderManager"            as mgr     <<AuthenticationManager>>
participant  "DriveThruAuthenticationProvider" as prov <<AuthenticationProvider>>
participant  "DriveThruRPGService"        as svc     <<@Service, @Cacheable>>
participant  "DriveThruRPGClient"         as http    <<RestClient Proxy>>
boundary     "DriveThruRPG API"           as api

client ->> filter : HTTP\n Authorization: ApiKey <key>
activate filter

filter ->> mgr    : authenticate(\n  unauthenticated(apiKey))
activate mgr

mgr ->> prov      : authenticate(token)
activate prov

prov ->> svc      : getToken(apiKey)
activate svc

svc ->> http      : getToken("Bearer " + apiKey)
activate http
http ->> api      : POST /token
api -->> http     : {access_token, customers_id, expires}
http -->> svc     :
deactivate http

svc -->> prov     : DrivethruToken
deactivate svc

prov -->> mgr     : ApiKeyAuthenticationToken\n  .authenticated(userDetails)
deactivate prov

mgr -->> filter   :
deactivate mgr

filter ->> filter : SecurityContextHolder.set(auth)
filter -->> client : 200 + request continues
deactivate filter

@enduml
----
